# ============================================
# CHARTVOLT ENVIRONMENT VARIABLES
# ============================================
# Copy this file to .env and fill in your values:
#   cp env_minimal_example.txt .env
#   nano .env
#
# The setup-new-customer.sh script will generate this automatically.

# ============================================
# NODE ENVIRONMENT
# ============================================
NODE_ENV=production

# ============================================
# APP URLS (Replace yourdomain.com with your actual domain)
# ============================================
NEXT_PUBLIC_BASE_URL=https://yourdomain.com
NEXT_PUBLIC_APP_URL=https://yourdomain.com
ADMIN_URL=https://admin.yourdomain.com

# ============================================
# MONGODB (Required - get from MongoDB Atlas)
# ============================================
MONGODB_URI=mongodb+srv://username:password@cluster.mongodb.net/chartvolt

# ============================================
# AUTHENTICATION (Required)
# ============================================
# Generate with: openssl rand -hex 32
BETTER_AUTH_SECRET=your-super-secret-key-min-32-chars
BETTER_AUTH_URL=https://yourdomain.com

# ============================================
# ADMIN PANEL (Required for first admin login)
# ============================================
ADMIN_EMAIL=admin@yourdomain.com
ADMIN_PASSWORD=change-this-secure-password
# Generate with: openssl rand -hex 32
ADMIN_JWT_SECRET=your-super-secret-admin-key-change-in-production

# ============================================
# API SERVER
# ============================================
API_PORT=4000

# ============================================
# WEBSOCKET SERVER (Real-time messaging)
# ============================================
WEBSOCKET_PORT=3003
NEXT_PUBLIC_WEBSOCKET_URL=wss://yourdomain.com/ws
WEBSOCKET_INTERNAL_URL=http://localhost:3003

# ============================================
# ALL OTHER SETTINGS
# ============================================
# The following settings are managed via the Admin Panel:
# - Redis connection (Host, Port, Password)
# - Nodemailer (Email) credentials
# - Gemini API Key (AI)
# - Massive API Keys (Forex data)
# - Payment Provider credentials (Stripe, Paddle, etc.)
#
# To configure these:
# 1. Start the app
# 2. Login to admin panel (https://admin.yourdomain.com)
# 3. Go to Settings > Environment Variables OR Settings > Payment Providers
# 4. Configure your credentials there
#
# Benefits of admin panel configuration:
# - No need to edit .env for most settings
# - Settings persist in database
# - Can be updated via UI without server restart
# - More secure and manageable
